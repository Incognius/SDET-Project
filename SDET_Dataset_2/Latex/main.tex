\begin{frame}[t]
\frametitle{Introduction}
\begin{itemize}
    \item The two-sample z/t-test examines whether two populations have the same mean, using their observed samples.
    \vspace{0.5em}
    \item \textbf{$H_{0}$:  $\mu_{x} = \mu_{y}$}
    \item \textbf{$H_{1}:$  $\mu_{x} \ne \mu_{y}$}
    \vspace{0.5em}
    \item Two different distributions can have the same population mean.
    \vspace{0.5em}
    \begin{enumerate}
        \item \(X\sim\mathcal{N}(0,1)\) and \(Y\sim\mathcal{N}(0,4)\)
        \vspace{0.3em}
        \item \(X\sim\mathcal{N}(2,1)\) and \(Y\sim\operatorname{Exp}(\lambda=1/2)\)
    \end{enumerate}
    \vspace{0.5em}
    \item But, in some cases we might have some prior knowledge of the samples and know that both belong to the same distribution family (ex: Gaussian), and have the same population variance.
    \vspace{0.3em}
    \begin{itemize}
        \item In this case, if $\mu_{x} = \mu_{y}$, both distributions are same - what we want to check.
    \end{itemize}
\end{itemize}
    
\end{frame}

\begin{frame}[t]
\begin{itemize}
    \item Even if two populations have the same mean, two randomly drawn samples will generally have different sample means due to sampling variability.
    \vspace{0.3em}
    \item We want to determine whether the observed difference between the sample means is statistically significant enough to reject \(H_0\).
    
\end{itemize}
    
\end{frame}



\begin{frame}[t]
\frametitle{When is a mean difference significant?}
\begin{itemize}
    \item First we measure how much the sample means differ: $\bar{X}-\bar{Y}$
    \vspace{0.3em}
    \item Let's say we get D = 5. 
    \vspace{0.1em}
    \begin{enumerate}
        \item \textit{Situation A}: Values usually between 49 \& 51, so D = 5 is relatively large.
        \vspace{0.2em}
        \item \textit{Situation B}: Values usually between 0 \& 100, so D = 5 may be relatively small.
    \end{enumerate}
    \vspace{0.3em}
    \item The observed difference alone is insufficient. We must account for the natural variability of the sample means.
    \vspace{0.3em}
    \item We quantify this variability using the \textbf{standard error.}

\end{itemize}
    
\end{frame}

\begin{frame}[t]
\frametitle{Standard Error}
\begin{itemize}
    \item \textbf{Standard deviation} measures how spread out individual observations are around their mean.
    \item However, we are comparing sample means, not individual observations.
    \item Even if the population means are equal, the sample means can differ due to random sampling.
    \item Standard error (SE) measures how much a sample statistic (such as the sample mean) would vary across repeated random samples.
 \[
\boxed{
SE(\bar X-\bar Y)=
\sqrt{\frac{s_X^2}{n_X}+\frac{s_Y^2}{n_Y}}
}
\]
\end{itemize}
    
\end{frame}

\begin{frame}[t]
\frametitle{Test Statistic}
\[
\boxed{
t=\frac{\bar X-\bar Y}{SE(\bar X-\bar Y)}
}
\]

\begin{itemize}
    \item The further away this value is from 0, the more likely it is that we can reject $H_{0}$
    \vspace{0.3em}
    \item But how do we determine whether the observed t-value is sufficiently extreme to reject \(H_0\)? We calculate its \textbf{p-value.}
    \vspace{0.3em}
    \item Assuming \(H_0\) is true, the p-value is the probability of observing a test statistic at least as extreme as the one obtained.
    \vspace{0.3em}
    \item We compare the p-value against a chosen significance level, \(\alpha\) (typically 0.05).
    \vspace{0.3em} \[
\boxed{
\begin{aligned}
p<\alpha &\implies \text{Reject }H_0\\
p\geq\alpha &\implies \text{Fail to reject }H_0
\end{aligned}
}
\]
\end{itemize}
    
\end{frame}


\begin{frame}[t]
\frametitle{Z-test}
\begin{itemize}
    \item The difference between z-test and t-test is that in z-test, the population variances are known.
\end{itemize}
\vspace{0.5em}
\[
\boxed{z=\frac{\bar X-\bar Y}{\sqrt{\frac{\sigma_X^2}{n_X}+\frac{\sigma_Y^2}{n_Y}}}}
\]
\end{frame}

\begin{frame}[t]
\frametitle{KS Test: Introduction}
\begin{itemize}
    \item The Kolmogorov--Smirnov (KS) test examines whether two populations have the same \textbf{distribution}, not just the same mean, using their observed samples.
    \vspace{0.5em}
    \item \textbf{$H_{0}$:  $F_{X} = F_{Y}$}
    \item \textbf{$H_{1}:$  $F_{X} \ne F_{Y}$}
    \vspace{0.5em}
    \item Here $F$ is the cumulative distribution function (CDF): $F(x)$ is the probability that a value is at most $x$.
    \vspace{0.5em}
    \item The t-test cannot separate \(X\sim\mathcal{N}(0,1)\) from \(Y\sim\mathcal{N}(0,4)\), since both have mean 0. The KS test can, because their distributions differ.
    \vspace{0.5em}
    \item The KS test is \textbf{nonparametric}: it does not assume that the data are Gaussian or that the variances are equal.
\end{itemize}
\end{frame}

\begin{frame}[t]
\frametitle{Empirical CDF}
\begin{itemize}
    \item We do not know the true CDF, so we estimate it from a sample by counting.
    \item The \textbf{empirical CDF} is the fraction of the sample that is at most $x$:
 \[
\boxed{
\hat F_X(x)=\frac{1}{n_X}\sum_{i=1}^{n_X}\mathbf{1}\{X_i\le x\}
}
\]
    \item Here $\mathbf{1}\{\cdot\}$ is 1 when the condition is true and 0 otherwise.
    \vspace{0.3em}
    \item Example: for $X=\{1,2,4,7\}$, $\hat F_X(3)=\frac{2}{4}=0.5$.
    \vspace{0.3em}
    \item It is a staircase: it starts at 0, rises by $1/n_X$ at each data point, and ends at 1.
    \item Even if $H_0$ is true, $\hat F_X$ and $\hat F_Y$ will differ slightly due to sampling variability.
\end{itemize}
\end{frame}

\begin{frame}[t]
\frametitle{KS Statistic}
\small
\[
\boxed{
D=\max_{x}\left|\hat F_X(x)-\hat F_Y(x)\right|
}
\]

\begin{itemize}
    \item $D$ is the largest vertical gap between the two empirical CDFs, so $0 \le D \le 1$.
    \item The further $D$ is from 0, the more likely it is that we can reject $H_{0}$.
    \item Both CDFs only change at data points, so it is enough to check the gap at the data points.
\end{itemize}

\centering
\includegraphics[width=0.62\textwidth]{ks_gap.png}
\end{frame}

\begin{frame}[t]
\frametitle{KS Test: p-value and Decision}
\begin{itemize}
    \item How large must $D$ be? We compare it with how large it would typically be if $H_0$ were true, just as the t-test compares the difference of means with the standard error.
    \vspace{0.3em}
    \item For continuous data, the distribution of $D$ under $H_0$ is the same whatever the data distribution is, so no Gaussian assumption is needed.
    \vspace{0.3em}
    \item For large samples, we reject $H_0$ at $\alpha = 0.05$ if
 \[
\boxed{
\sqrt{\frac{n_X n_Y}{n_X+n_Y}}\;D > 1.358
}
\]
    \item The factor $\sqrt{n_X n_Y/(n_X+n_Y)}$ rescales $D$ for the sample sizes, similar to dividing by the standard error in the t-test.
    \vspace{0.3em}
    \item For small samples the exact distribution of $D$ is used. The p-value is the probability, assuming $H_0$ is true, of observing a $D$ at least as large as the one obtained, and the decision rule is the same: reject $H_0$ if $p<\alpha$.
\end{itemize}
\end{frame}

\begin{frame}[t]
\frametitle{What the KS Test Can See}
\begin{itemize}
    \item The KS test reacts to \textbf{any} change in the distribution: a shift in the mean, a change in the variance, or a change in shape.
    \vspace{0.3em}
    \item The t-test reacts only to a change in the mean. If the mean is unchanged, it cannot see a variance shift.
    \vspace{0.3em}
    \item The price of this generality: for a pure mean shift in Gaussian data, the t-test is the more powerful test, since it uses exactly the information that matters.
    \vspace{0.3em}
    \item In our experiment, the KS test is applied to each of the 10 features separately, and we report a shift if \textbf{any} feature has $p<0.05$ (the same rule as the t-test).
\end{itemize}

\end{frame}

\begin{frame}[t]
\frametitle{GLRT: Introduction}
\begin{itemize}
    \item The generalised likelihood ratio test (GLRT) asks: how much better do the data fit if each sample gets its \textbf{own} Gaussian, instead of one shared Gaussian?
    \vspace{0.4em}
    \item \textbf{$H_0$:} $(\mu_X,\Sigma_X)=(\mu_Y,\Sigma_Y)$ \quad (one Gaussian for both samples)
    \item \textbf{$H_1$:} $(\mu_X,\Sigma_X)\ne(\mu_Y,\Sigma_Y)$ \quad (a separate Gaussian for each)
    \vspace{0.4em}
    \item The likelihood ratio compares the best fit under each hypothesis:
\[
\Lambda=\frac{\max_{H_0} L}{\max_{H_1} L}\le 1,
\qquad
\boxed{G=-2\log\Lambda}
\]
    \item $\Lambda$ near 1 (G near 0): one Gaussian fits as well as two, so no evidence of a shift. Large $G$: reject $H_0$.
    \item Unlike the t-test, it sees mean \textbf{and} covariance changes.
\end{itemize}
\end{frame}

\begin{frame}[t]
\frametitle{Step 1: Write the Gaussian likelihood}
\small
For one observation $x\in\mathbb{R}^d$:
\[
p(x\mid\mu,\Sigma)=\frac{1}{(2\pi)^{d/2}|\Sigma|^{1/2}}
\exp\!\left[-\tfrac12(x-\mu)^\top\Sigma^{-1}(x-\mu)\right]
\]
\begin{itemize}
    \item $(2\pi)^{-d/2}$ is a dimensional normalisation factor.
    \item $|\Sigma|^{-1/2}$ accounts for the volume and spread of the Gaussian.
    \item The exponential penalises observations far from the mean in covariance-adjusted distance.
\end{itemize}
For independent observations, likelihoods multiply, so log-likelihoods add. For $N$ observations $z_1,\dots,z_N$:
\[
\ell(\mu,\Sigma)=-\frac{Nd}{2}\log(2\pi)-\frac N2\log|\Sigma|
-\frac12\sum_{i=1}^{N}(z_i-\mu)^\top\Sigma^{-1}(z_i-\mu)
\]
This is our starting point.
\end{frame}

\begin{frame}[t]
\frametitle{Step 2: Maximum-likelihood estimates}
\small
\[
\boxed{\hat\mu=\bar z=\frac1N\sum_{i=1}^N z_i}
\qquad
\boxed{\hat\Sigma=\frac1N\sum_{i=1}^N(z_i-\bar z)(z_i-\bar z)^\top}
\]
\begin{itemize}
    \item The denominator is $N$, not $N-1$: this is the Gaussian MLE, not the unbiased sample covariance.
\end{itemize}
Substituting the fitted parameters, the quadratic terms collapse (using $a^\top Ba=\operatorname{tr}(Baa^\top)$):
\[
\sum_{i=1}^N(z_i-\bar z)^\top\hat\Sigma^{-1}(z_i-\bar z)
=\operatorname{tr}\!\Big[\hat\Sigma^{-1}\textstyle\sum_i(z_i-\bar z)(z_i-\bar z)^\top\Big]
=\operatorname{tr}\big[\hat\Sigma^{-1}N\hat\Sigma\big]=Nd
\]
So the maximised log-likelihood is
\[
\boxed{\hat\ell=-\frac N2\Big[d\log(2\pi)+d+\log|\hat\Sigma|\Big]}
\]
After fitting, the likelihood depends on the data \textbf{only through} $\log|\hat\Sigma|$.
\end{frame}

\begin{frame}[t]
\frametitle{Steps 3--4: Likelihood under $H_0$ and $H_1$}
\small
\textbf{Under $H_0$}, pool all $N=n+m$ observations and fit one Gaussian:
\[
\bar z=\frac1N\sum_{i=1}^N z_i,\qquad
\hat\Sigma_0=\frac1N\sum_{i=1}^N(z_i-\bar z)(z_i-\bar z)^\top
\]
\[
\hat\ell_0=-\frac N2\Big[d\log(2\pi)+d+\log|\hat\Sigma_0|\Big]
\]
\begin{itemize}
    \item $\hat\Sigma_0$ is taken about the grand mean, so a mean difference between $X$ and $Y$ shows up as \textbf{extra spread}.
\end{itemize}
\textbf{Under $H_1$}, fit $X$ and $Y$ separately; their log-likelihoods add:
\[
\hat\ell_1=-\frac n2\Big[d\log(2\pi)+d+\log|\hat\Sigma_X|\Big]
-\frac m2\Big[d\log(2\pi)+d+\log|\hat\Sigma_Y|\Big]
\]
\end{frame}

\begin{frame}[t]
\frametitle{Step 5: The test statistic}
Since $\log\Lambda=\hat\ell_0-\hat\ell_1$, we obtain $G=2(\hat\ell_1-\hat\ell_0)$.

\vspace{0.3em}
Substituting the fitted log-likelihoods, all constant terms cancel ($N=n+m$), leaving
\[
\boxed{G=N\log|\hat\Sigma_0|-n\log|\hat\Sigma_X|-m\log|\hat\Sigma_Y|}
\]
\begin{itemize}
    \item This is the Gaussian likelihood-ratio statistic for testing equality of two multivariate Gaussians.
    \vspace{0.3em}
    \item Intuition: if $X$ and $Y$ come from the same Gaussian, pooling costs nothing and $|\hat\Sigma_0|$ is close to the separate determinants. A shift in mean or covariance inflates $|\hat\Sigma_0|$, so $G$ grows.
    \vspace{0.3em}
    \item \textbf{Diagonal case} (features assumed independent, as in our experiment): $|\Sigma|=\prod_j\sigma_j^2$, so
\[
G=\sum_{j=1}^{d}\Big[N\log\hat\sigma^2_{0j}-n\log\hat\sigma^2_{Xj}-m\log\hat\sigma^2_{Yj}\Big]
\]
\end{itemize}
\end{frame}

\begin{frame}[t]
\frametitle{Reference distribution and dimensionality}
\small
\textbf{Wilks' theorem:} under $H_0$ and for large $N$, $G\approx\chi^2_r$, where $r$ is the number of extra free parameters $H_1$ has over $H_0$.
\vspace{0.3em}

\centering
\begin{tabular}{l|c|c|c}
Model & params per Gaussian & $r$ & $d=10$ \\ \hline
Full covariance & $d+\frac{d(d+1)}{2}$ & $d+\frac{d(d+1)}{2}$ & 65 \\
Diagonal & $d+d$ & $2d$ & \textbf{20} \\
\end{tabular}

\raggedright
\vspace{0.4em}
\begin{itemize}
    \item The diagonal model is why our test uses $\chi^2_{20}$.
    \item \textbf{Dimensionality cost:} the full model estimates $\sim d^2/2$ covariance entries per sample.
    \begin{itemize}
        \item $\hat\Sigma_X$ has rank $\le n-1$, so $|\hat\Sigma_X|=0$ and $G$ is undefined unless $n>d$.
        \item Wilks needs $N\gg r$. With $r=65$ and $n=20$ the $\chi^2$ approximation is poor.
    \end{itemize}
    \item Wilks also assumes the Gaussian model is \textbf{correct}. Heavy tails break it, which is why the heavy-tailed false alarms reach 93\%.
    \item Fix: keep $G$ but get its null distribution by \textbf{permutation} (shuffle $X$/$Y$ labels, recompute $G$).
\end{itemize}
\end{frame}

    \item The GLRT requires a suitable distributional model for its reference distribution to be valid.
\end{itemize}
